from opentrons import protocol_api

metadata = {
    'protocolName': 'BioNeMo NIM KRAS G12D Assay Dispensing',
    'author': 'Aarav Jorwal (@Jorwalzzz)',
    'apiLevel': '2.15'
}

def run(protocol: protocol_api.ProtocolContext):
    # 1. Load Deck Hardware
    plate = protocol.load_labware('corning_96_wellplate_360ul_flat', 1)
    tiprack = protocol.load_labware('opentrons_96_tiprack_300ul', 2)
    p300 = protocol.load_instrument('p300_single_gen2', 'right', tip_racks=[tiprack])

    # 2. Automated Mastermix & Inhibitor Serial Dilution
    p300.transfer(50, plate.wells()[0], plate.wells()[1:12], mix_after=(3, 30))
    protocol.comment("OT-2 BioNeMo dispensing completed.")
